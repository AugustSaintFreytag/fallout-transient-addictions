; Aux Vars

; Aux vars do not support string-keyed hash maps, so a pair of key/value arrays will be used.
; "LastActivationMapKeys" ; Array of effect name strings
; "LastActivationMapValues" ; Array of days passed since last activation
; "*RecoveryMessageQueue" ; Array of item names to be shown in recovery messages
; "*RecoveryMessages" ; Array of cached message texts read from INI

; UDFs

ref OnItemActivationUDF = CompileScript "SaintsTransientAddictions\Items\OnItemActivation.gek"
ref TickAllPlayerEffectsUDF = CompileScript "SaintsTransientAddictions\Tick\TickAllPlayerEffects.gek"

; Config

short iTickIntervalSeconds = GetINIFloat_Cached "General:iTickInterval" "SaintsTransientAddictions.ini" 15
short iTickIntervalFrames = iTickIntervalSeconds * 60

; Event Handlers

SetEventHandlerAlt "ShowOff:OnPreActivateInventoryItem" (begin function { ref rBaseItem, ref rInvRef, short bIsActivationNotPrevented, short iActivatedHotkey } 
	Call OnItemActivationUDF rBaseItem
end)

SetEventHandlerAlt "QuickSelect:OnPreActivateInventoryItem" (begin function { ref rBaseItem, array_var aEventState } 
	Call OnItemActivationUDF rBaseItem
end)

SetGameMainLoopCallback TickAllPlayerEffectsUDF 1 iTickIntervalFrames 1

Print "Initialized Saint's Transient Addictions event handlers and tick loop."
